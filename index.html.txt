    // LÓGICA DE CARTERA COLOMBIANA: 5 DÍAS HÁBILES DE GRACIA (EXCLUYE FINES DE SEMANA)
    function calcularEstadoMora(u) {
      if (!u || !u.ocupado || u.pagado) {
        return { enMora: false, diasMora: 0, fechaLimite: '', diaCorte: 5, fechaCorteStr: '' };
      }

      const ahora = new Date();
      const anioActual = ahora.getFullYear();
      const mesActual = ahora.getMonth(); // 0-indexed
      const hoySinHora = new Date(anioActual, mesActual, ahora.getDate());

      let diaCorte = 5;
      let fechaInicioContrato = null;

      if (u.fechaArrendamiento && typeof u.fechaArrendamiento === 'string' && u.fechaArrendamiento.includes('-')) {
        const partes = u.fechaArrendamiento.split('-');
        const y = parseInt(partes[0], 10);
        const m = parseInt(partes[1], 10) - 1;
        const d = parseInt(partes[2], 10);
        diaCorte = d || 5;
        fechaInicioContrato = new Date(y, m, d);
      } else if (u.diaPago) {
        diaCorte = parseInt(u.diaPago, 10) || 5;
      }

      // Determinar la fecha de vencimiento relevante para el periodo impago:
      // Si hoy ya pasó o es igual al día de corte en este mes, el corte a evaluar es de este mes.
      // Si hoy es ANTERIOR al día de corte (por ej. hoy es 2 de nov y el corte es el 23),
      // significa que el arriendo impago corresponde al corte del mes anterior (23 de oct),
      // siempre y cuando el contrato ya estuviera vigente en esa fecha.
      const diasEnEsteMes = new Date(anioActual, mesActual + 1, 0).getDate();
      const diaEfectivoEsteMes = Math.min(diaCorte, diasEnEsteMes);
      const fechaCorteEsteMes = new Date(anioActual, mesActual, diaEfectivoEsteMes);

      let fechaCorteEvaluada = null;

      if (hoySinHora >= fechaCorteEsteMes) {
        fechaCorteEvaluada = fechaCorteEsteMes;
      } else {
        // El corte en este mes aún no llega. Evaluamos el corte del mes anterior:
        let anioAnt = anioActual;
        let mesAnt = mesActual - 1;
        if (mesAnt < 0) { mesAnt = 11; anioAnt--; }
        const diasEnMesAnt = new Date(anioAnt, mesAnt + 1, 0).getDate();
        const diaEfectivoAnt = Math.min(diaCorte, diasEnMesAnt);
        const fechaCorteMesAnt = new Date(anioAnt, mesAnt, diaEfectivoAnt);

        if (fechaInicioContrato && fechaInicioContrato > fechaCorteMesAnt) {
          fechaCorteEvaluada = fechaInicioContrato;
        } else {
          fechaCorteEvaluada = fechaCorteMesAnt;
        }
      }

      // Si el inicio de contrato es posterior a la fecha evaluada, usamos el inicio del contrato
      if (fechaInicioContrato && fechaInicioContrato > fechaCorteEvaluada) {
        fechaCorteEvaluada = fechaInicioContrato;
      }

      // Calcular fecha límite sumando exactamente 5 días hábiles (Lunes a Viernes)
      let fechaLimite = new Date(fechaCorteEvaluada.getTime());
      let diasHabilesSumados = 0;
      while (diasHabilesSumados < 5) {
        fechaLimite.setDate(fechaLimite.getDate() + 1);
        const dayOfWeek = fechaLimite.getDay(); // 0: domingo, 6: sábado
        if (dayOfWeek !== 0 && dayOfWeek !== 6) {
          diasHabilesSumados++;
        }
      }

      // Comparar hoy con la fecha límite de gracia
      const enMora = hoySinHora > fechaLimite;
      let diasMora = 0;
      if (enMora) {
        const diffMs = hoySinHora.getTime() - fechaLimite.getTime();
        diasMora = Math.floor(diffMs / (1000 * 60 * 60 * 24));
      }

      return {
        enMora: enMora,
        diasMora: diasMora,
        fechaLimite: fechaLimite.toLocaleDateString('es-CO'),
        fechaCorteStr: fechaCorteEvaluada.toLocaleDateString('es-CO'),
        diaCorte: diaCorte
      };
    }

    // HELPER PARA CALCULAR EL PERIODO DE ARRENDAMIENTO PAGADO (DESDE QUÉ FECHA HASTA QUÉ FECHA)
    function calcularPeriodoArriendo(u, fechaRefStr) {
      let diaCorte = 5;
      let anioContrato = 2026;
      let mesContrato = 10; // 0-indexed

      if (u.fechaArrendamiento && typeof u.fechaArrendamiento === 'string' && u.fechaArrendamiento.includes('-')) {
        const p = u.fechaArrendamiento.split('-');
        anioContrato = parseInt(p[0], 10);
        mesContrato = parseInt(p[1], 10) - 1;
        diaCorte = parseInt(p[2], 10) || 5;
      } else if (u.diaPago) {
        diaCorte = parseInt(u.diaPago, 10) || 5;
      }

      // Fecha de referencia: fecha de pago asentada o fecha actual
      let fechaRef = new Date();
      if (fechaRefStr && typeof fechaRefStr === 'string' && fechaRefStr.includes('-')) {
        const fp = fechaRefStr.split('-');
        fechaRef = new Date(parseInt(fp[0], 10), parseInt(fp[1], 10) - 1, parseInt(fp[2], 10));
      } else if (u.fechaPagoMensual && u.fechaPagoMensual.includes('-')) {
        const fp = u.fechaPagoMensual.split('-');
        fechaRef = new Date(parseInt(fp[0], 10), parseInt(fp[1], 10) - 1, parseInt(fp[2], 10));
      }

      const anioRef = fechaRef.getFullYear();
      const mesRef = fechaRef.getMonth();
      const diaRef = fechaRef.getDate();

      let startYear = anioRef;
      let startMonth = mesRef;

      if (diaRef < diaCorte) {
        startMonth = mesRef - 1;
        if (startMonth < 0) {
          startMonth = 11;
          startYear--;
        }
      }

      // Validar si el contrato comenzó después de esa fecha
      const fechaInicioContrato = new Date(anioContrato, mesContrato, diaCorte);
      const diasEnMesInicio = new Date(startYear, startMonth + 1, 0).getDate();
      const diaEfectivoInicio = Math.min(diaCorte, diasEnMesInicio);
      let startDate = new Date(startYear, startMonth, diaEfectivoInicio);

      if (startDate < fechaInicioContrato) {
        startDate = fechaInicioContrato;
      }

      // Fecha fin: un mes después menos 1 día
      let endMonth = startDate.getMonth() + 1;
      let endYear = startDate.getFullYear();
      if (endMonth > 11) {
        endMonth = 0;
        endYear++;
      }
      const diasEnMesFin = new Date(endYear, endMonth + 1, 0).getDate();
      const diaEfectivoFin = Math.min(startDate.getDate(), diasEnMesFin);
      const nextMonthDue = new Date(endYear, endMonth, diaEfectivoFin);
      const endDate = new Date(nextMonthDue.getTime() - (24 * 60 * 60 * 1000));

      const mesesNombres = [
        "Enero", "Febrero", "Marzo", "Abril", "Mayo", "Junio",

    function renderDashboard() {
      const grid = document.getElementById('propsList');
      grid.innerHTML = '';
      let totalIngresos = 0;
      let totalMora = 0;
      const diaHoy = new Date().getDate();

      let totalSinPagar = 0;

      CONFIG_PROPS.forEach(p => {
        let occ = 0, cob = 0, pen = 0;
        for (let i = 1; i <= p.tot; i++) {
          const u = db.units[`${p.id}_${i}`];
          if (u.ocupado) {
            occ++;
            if (u.pagado) {
              cob++;
              totalIngresos += Number(u.canon || 0);
            } else {
              pen++;
              totalSinPagar++;
              const mInfo = calcularEstadoMora(u);
              if (mInfo.enMora) totalMora++;
            }
          }
        }
        grid.innerHTML += `
          <div class="prop-box" onclick="abrirProp('${p.id}')">
            <div class="prop-top">
              <div class="prop-ico" style="background:${p.ico};">
                <svg width="18" height="18" fill="white" viewBox="0 0 24 24"><path d="M10 20v-6h4v6h5v-8h3L12 3 2 12h3v8z"/></svg>
              </div>
              <span class="prop-badge">${occ}/${p.tot} Occup</span>
            </div>
            <div class="prop-name">${p.name}</div>
            <div class="prop-sub">${p.sub}</div>
            <div class="prop-foot">
              <span style="color:#056648;">${cob} Cobrados</span>
              <span style="color:var(--danger); font-weight:800;">${pen} Sin pagar</span>
            </div>
          </div>
        `;
      });

      const totalGastos = db.gastos.reduce((acc, g) => acc + Number(g.monto || 0), 0);
      const gananciaNeta = totalIngresos - totalGastos;

      document.getElementById('txtIngresos').innerText = formatCOP(totalIngresos);
      document.getElementById('txtGastos').innerText = formatCOP(totalGastos);
      document.getElementById('txtNeto').innerText = formatCOP(gananciaNeta);
      document.getElementById('valGastosResumen').innerText = formatCOP(totalGastos);
      document.getElementById('lblGastosMes').innerText = `${db.gastos.length} gastos en este mes`;

      const mBanner = document.getElementById('moraBanner');
      const mTitulo = document.getElementById('moraTitulo');
      const mSub = document.getElementById('moraSub');
      const mIcon = document.getElementById('moraIcon');
      const mBtn = document.getElementById('moraBtnDetalles');
      const tagAlertas = document.getElementById('tagAlertas');

    function toggleRecibo(uid) {
      const u = db.units[uid];
      if (!u.pagado) {
        const ref = prompt("Ingrese comprobante o referencia de pago:", "Nequi / Consignación");
        if (ref) {
          u.pagado = true;
          u.soporte = ref;
          u.fechaPagoReg = new Date().toLocaleDateString('es-CO');
          if (!u.fechaPagoMensual) {
            u.fechaPagoMensual = new Date().toISOString().slice(0, 10);
          }
        }
      } else {
        if (confirm(`El pago ya está registrado con soporte: "${u.soporte}". ¿Marcar nuevamente como Pendiente?`)) {
          u.pagado = false;
          u.soporte = '';
        u.soporteImg = '';
          u.fechaPagoReg = '';
          u.fechaPagoMensual = '';
        }
      }
      saveDB();
      if (document.getElementById('modalProp').classList.contains('active')) {
        abrirProp(u.propId);
      }
    }

    function procesarArchivoRecibo(event) {
      const file = event.target.files && event.target.files[0];
      if (!file || !unidadSeleccionadaParaFoto) return;

      const uid = unidadSeleccionadaParaFoto;
      const u = db.units[uid];
      if (!u) return;

      const reader = new FileReader();
      if (file.type.startsWith('image/')) {
        // Redimensionar para optimizar peso y evitar saturar Google Sheets / LocalStorage
        reader.onload = function(e) {
          const img = new Image();
          img.onload = function() {
            const canvas = document.createElement('canvas');
            let width = img.width;
            let height = img.height;
            const maxDim = 900;
            if (width > height && width > maxDim) {
              height = Math.round((height * maxDim) / width);
              width = maxDim;
            } else if (height > maxDim) {
              width = Math.round((width * maxDim) / height);
              height = maxDim;
            }
            canvas.width = width;
            canvas.height = height;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, width, height);
            const dataUrlOptimizada = canvas.toDataURL('image/jpeg', 0.72);

            u.pagado = true;
            u.soporte = file.name;
            u.soporteImg = dataUrlOptimizada;
            u.fechaPagoReg = new Date().toLocaleDateString('es-CO');
            u.fechaPagoMensual = new Date().toISOString().slice(0, 10);
            saveDB();
            if (document.getElementById('modalProp').classList.contains('active')) {
              abrirProp(u.propId);
            }
            setTimeout(() => { generarReciboPagoPDF(uid); }, 400);
          };
          img.src = e.target.result;
        };
        reader.readAsDataURL(file);
      } else {
        // Para PDF u otros archivos compatibles
        reader.onload = function(e) {
          u.pagado = true;
          u.soporte = file.name;
          u.soporteImg = e.target.result;
          u.fechaPagoReg = new Date().toLocaleDateString('es-CO');
          u.fechaPagoMensual = new Date().toISOString().slice(0, 10);
          saveDB();
          if (document.getElementById('modalProp').classList.contains('active')) {
            abrirProp(u.propId);
          }
          setTimeout(() => { generarReciboPagoPDF(uid); }, 400);
        };
        reader.readAsDataURL(file);
      }
    }

        inputF.value = new Date().toISOString().slice(0, 10);
      }

      let subPereira = 0;
      let subCombia = 0;

      db.gastos.forEach(g => {
        const esCombia = g.inmueble === 'Gastos Combia' || (g.inmueble && g.inmueble.toLowerCase().includes('combia'));
        if (esCombia) subCombia += Number(g.monto || 0);
        else subPereira += Number(g.monto || 0);
      });

      const pEl = document.getElementById('txtModalGastosPereira');
      const cEl = document.getElementById('txtModalGastosCombia');
      if (pEl) pEl.innerText = formatCOP(subPereira);
      if (cEl) cEl.innerText = formatCOP(subCombia);

      if (db.gastos.length === 0) {
        c.innerHTML = '<div style="font-size:0.75rem; color:var(--subtext); text-align:center; padding:10px;">Sin gastos registrados.</div>';
      } else {
        db.gastos.slice().reverse().forEach((g, idxRev) => {
          const originalIdx = db.gastos.length - 1 - idxRev;
          const esCombia = g.inmueble === 'Gastos Combia' || (g.inmueble && g.inmueble.toLowerCase().includes('combia'));
          c.innerHTML += `
            <div style="display:flex; justify-content:space-between; align-items:center; padding:8px 0; border-bottom:1px solid #f1f5f9;">
              <div>
                <div style="font-size:0.82rem; font-weight:700;">${g.concepto}</div>
                <div style="font-size:0.68rem; color:var(--subtext); margin-top:2px;">
                  ?? ${g.fecha || 'Fecha no registrada'} &bull; 
                  <span style="font-weight:700; color:${esCombia ? '#d97706' : '#056648'};">
                    ${esCombia ? '?? Gastos Combia' : '?? ' + g.inmueble}
                  </span>
                </div>
              </div>
              <div style="text-align:right;">
                <div style="font-size:0.85rem; font-weight:800; color:var(--danger);">${formatCOP(g.monto)}</div>
                <button onclick="borrarGasto(${originalIdx})" style="border:none; background:none; color:#dc2626; font-size:0.68rem; cursor:pointer; font-weight:700;">Eliminar</button>
              </div>
            </div>
          `;
        });
      }
      abrirModal('modalGastos');
    }

    function guardarGasto() {
      const con = document.getElementById('gConcepto').value.trim();
      const mon = Number(document.getElementById('gMonto').value);
      const inm = document.getElementById('gInmueble').value;
      const fec = document.getElementById('gFecha').value || new Date().toISOString().slice(0, 10);
      if (!con || !mon || mon <= 0) return alert("Por favor complete el concepto y un valor válido.");

      db.gastos.push({
        concepto: con,
        monto: mon,
        inmueble: inm,
        fecha: fec
      });

      document.getElementById('gConcepto').value = '';
      document.getElementById('gMonto').value = '';
      saveDB();
      abrirGastos();
    }

    function borrarGasto(idx) {
      db.gastos.splice(idx, 1);
      saveDB();
      if (document.getElementById('modalGastos').classList.contains('active')) {
        abrirGastos();
      }
    }

    function exportarCSV() {
      let csv = "TIPO,INMUEBLE,DETALLE_O_INQUILINO,CEDULA,VALOR,ESTADO,FECHA_ARRENDAMIENTO,FECHA_PAGO_CONSIGNADA,PERIODO_DESDE,PERIODO_HASTA\n";
      Object.values(db.units).forEach(u => {
        const conf = CONFIG_PROPS.find(p => p.id === u.propId);
        const pi = calcularPeriodoArriendo(u, u.fechaPagoMensual); csv += `Arriendo,${conf.name} - ${u.name},${u.inquilino || 'Vacio'},${u.cedula || ''},${u.canon},${u.pagado ? 'Cobrado' : 'Pendiente'},${u.fechaArrendamiento || ''},${u.fechaPagoMensual || ''},${pi.desdeCorto},${pi.hastaCorto}\n`;
      });
      db.gastos.forEach(g => {
        csv += `Gasto,${g.inmueble},${g.concepto},${g.monto},Pagado,,${g.fecha || ""}\n`;
      });
      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
      const a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = `Liquidacion_Inmobiliaria_Inez_${new Date().toISOString().slice(0,10)}.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
    }
